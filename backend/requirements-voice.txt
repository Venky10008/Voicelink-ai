# ── Voice cloning engines (OPTIONAL, heavy) ───────────────────────────────────
# ~2 GB of downloads (torch + CUDA wheels + transformers). Only needed if you
# want REAL local voice cloning. Without this file everything else still works:
# text chat (Groq), Edge TTS replies, and Whisper speech-to-text.
#
#   pip install -r requirements-voice.txt
#
# ── transformers MUST be >= 5.3.0 ─────────────────────────────────────────────
# Do NOT downgrade transformers to 4.x and do NOT leave it at 5.2.0.
#
# Why: omnivoice 0.2.1 declares `transformers>=5.3.0`, and BOTH omnivoice and
# coqui-tts import `HiggsAudioV2TokenizerModel` from transformers. That symbol
# was introduced in transformers 5.3.0, so on 5.2.0 both engines raise
#     ImportError: cannot import name 'HiggsAudioV2TokenizerModel'
# and the app silently reports "Voice cloning engine not available" in the UI.
#
# Pinning exactly (rather than >=) keeps pip from resolving a version one of the
# two engines has not been tested against.
transformers==5.18.0

# Preferred engine: 600+ languages, Apache 2.0.
omnivoice==0.2.1

# Fallback engine: ~17 languages, CPML license.
# NOTE: needs FFmpeg + torchcodec for audio I/O on torch >= 2.9:
#   pip install coqui-tts[codec]      (or)      pip install torchcodec
# voice_clone_engine.py patches XTTS to use soundfile instead, but coqui-tts
# 0.27.5 raises at IMPORT time without torchcodec, so install it if you need it.
coqui-tts==0.27.5

# GPU support (NVIDIA). Skip these two lines for a CPU-only install.
# torch>=2.4
# torchaudio>=2.4
